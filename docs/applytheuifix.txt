fn convert_svg_to_xvg(svg_path: &Path) -> anyhow::Result<File> {
    let svg_data = std::fs::read_to_string(svg_path)?;
    let tree = usvg::Tree::from_str(&svg_data, &usvg::Options::default())?;

    let mut xvg_file = File::default();
    xvg_file.header.width  = tree.size.width() as u16;
    xvg_file.header.height = tree.size.height() as u16;

    // 1. Walk SVG paths and export Bézier curves
    for node in tree.root.descendants() {
        if let usvg::NodeKind::Path(p) = &*node.borrow() {
            let mut bez = BezPath::new();
            for seg in &p.data {
                match seg {
                    usvg::PathSegment::MoveTo(p) => bez.move_to((p.x, p.y)),
                    usvg::PathSegment::LineTo(p) => bez.line_to((p.x, p.y)),
                    usvg::PathSegment::QuadTo(p, c) => bez.quad_to((p.x, p.y), (c.x, c.y)),
                    usvg::PathSegment::CubicTo(p, c1, c2) => {
                        bez.curve_to((p.x, p.y), (c1.x, c1.y), (c2.x, c2.y))
                    }
                    usvg::PathSegment::ClosePath => bez.close_path(),
                }
            }

            let tf = p.transform.to_cols_array();
            xvg_file.paths.push(PathRecord {
                data: bincode::serialize(&bez).unwrap(),
                tf,
            });
        }
    }

    // 2. Store original SVG as asset (optional)
    xvg_file.assets.push(Asset {
        ty: AssetType::ImageSvg,
        name: svg_path.file_name().unwrap().to_string_lossy().into(),
        data: svg_data.into_bytes(),
        compressed: false,
        ..Default::default()
    });

    Ok(xvg_file)
}
